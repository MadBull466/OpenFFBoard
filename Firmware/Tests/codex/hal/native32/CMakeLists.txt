cmake_minimum_required(VERSION 3.16)
project(P12Native32 LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(FW "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(P12_DEVICE F407VG CACHE STRING "Target source/header set")
if(NOT P12_DEVICE MATCHES "^(F407VG|F407VG_DISCO|F411RE)$")
    message(FATAL_ERROR "Unsupported P12_DEVICE: ${P12_DEVICE}")
endif()
set(DRIVERS "${FW}/Targets/${P12_DEVICE}/Drivers")
set(P12_NATIVE_SOURCE_DIR "${DRIVERS}/STM32F4xx_HAL_Driver/Src" CACHE PATH "Real source directory, scratch mutations only")
option(P12_FULL_ASSERT "Enable the supplied HAL parameter assertions" OFF)
option(P12_UBSAN "Run the real undefined-behavior sanitizer independently of access callbacks" OFF)
option(P12_COVERAGE "GCC execution coverage" ON)
option(P12_ACCESS_MODEL "Instrument C MMIO accesses through the owned bus model" ON)
if(NOT P12_ACCESS_MODEL)
    message(FATAL_ERROR "The full native suite requires the explicit access model; raw RAM is not a peripheral model")
endif()
option(P12_FULL_LL "Include supplied optional LL ADC C APIs" ON)
file(GLOB HAL_SOURCES CONFIGURE_DEPENDS "${P12_NATIVE_SOURCE_DIR}/*.c")
file(GLOB CASES CONFIGURE_DEPENDS cases/test_*.cpp)
add_executable(p12_native32 ${HAL_SOURCES} ${CASES} host/native.cpp host/access_probe.c host/empty_control.c ../test_main.cpp)
target_compile_options(p12_native32 PRIVATE -m32 -include "${CMAKE_CURRENT_SOURCE_DIR}/host/cpu_boundary.h" -Wall -Wextra)
target_link_options(p12_native32 PRIVATE -m32)
if(P12_ACCESS_MODEL)
    if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "P12_ACCESS_MODEL requires the installed Clang compiler")
    endif()
    target_compile_options(p12_native32 PRIVATE "$<$<COMPILE_LANGUAGE:C>:-fsanitize=kernel-address;-mllvm=-asan-instrumentation-with-call-threshold=0;-mllvm=-asan-globals=0;-mllvm=-asan-stack=0;-mllvm=-asan-opt=0;-mllvm=-asan-opt-same-temp=0;-mllvm=-asan-optimize-callbacks=0>")
    target_compile_definitions(p12_native32 PRIVATE P12_ACCESS_MODEL)
    set(P12_BUS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/host/access.cpp" CACHE FILEPATH "Owned bus source, scratch mutations only")
    target_sources(p12_native32 PRIVATE "${P12_BUS_SOURCE}")
endif()
target_compile_features(p12_native32 PRIVATE c_std_11 cxx_std_20)
if(P12_FULL_ASSERT)
    target_compile_definitions(p12_native32 PRIVATE USE_FULL_ASSERT)
endif()
if(P12_FULL_LL)
    target_compile_definitions(p12_native32 PRIVATE USE_FULL_LL_DRIVER)
endif()
target_include_directories(p12_native32 PRIVATE host "${CMAKE_CURRENT_SOURCE_DIR}/.."
    "${DRIVERS}/STM32F4xx_HAL_Driver/Inc" "${DRIVERS}/CMSIS/Device/ST/STM32F4xx/Include"
    "${DRIVERS}/CMSIS/Include" "${FW}/Targets/${P12_DEVICE}/Core/Inc")
target_include_directories(p12_native32 PRIVATE "${FW}/Tests/support")
target_include_directories(p12_native32 SYSTEM PRIVATE "${FW}/Tests/third_party")
if(P12_DEVICE STREQUAL "F411RE")
    target_compile_definitions(p12_native32 PRIVATE STM32F411xE)
else()
    target_compile_definitions(p12_native32 PRIVATE STM32F407xx)
endif()
if(P12_COVERAGE)
    target_compile_options(p12_native32 PRIVATE --coverage -O0 -g)
    target_link_options(p12_native32 PRIVATE --coverage)
endif()
if(P12_UBSAN)
    target_compile_options(p12_native32 PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
    target_link_options(p12_native32 PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
enable_testing()
add_test(NAME p12_native32 COMMAND p12_native32 --no-colors)
set_tests_properties(p12_native32 PROPERTIES TIMEOUT 30)

find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME p12_native32_evidence_controls COMMAND ${Python3_EXECUTABLE} -B "${CMAKE_CURRENT_SOURCE_DIR}/test_evidence.py")
